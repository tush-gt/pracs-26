FOR WINDOWS
1. python --version
2. pip install numpy, pandas etc etc
3. create virtual env
4. python -m venv venv
5. .\venv\Scripts\Activate.ps1
6. pip install pandas numpy matplotlib scipy scikit-learn sklearn seaborne  

FOR UBUNTU
1. python3 --version
2. sudo apt update
3. sudo apt install python3 python3-pip python3-venv -y
4. python3 -m venv venv
5. source venv/bin/activate
6. python -m pip install numpy pandas matplotlib scipy scikit-learn
